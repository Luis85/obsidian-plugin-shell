import { describe, expect, it, vi } from 'vitest';
import { createServices } from '../../src/bootstrap/services';
import { createNoteFeatures } from '../../src/application/note-feature';
import { markdownCodec } from '../../src/infrastructure/markdown';
import { taskFeature } from '../fixtures/tasks/definition';
import { projectFeature } from '../fixtures/projects/definition';
import { browserDocumentStorage } from '../../harness/app/document-storage';
import { host } from './helpers';

describe('Real runtime repository composition', () => {
  it('registers Task and Project with one storage contract, preferences folder and independent definitions', async () => {
    let files: Record<string, string> = {}; let sequence = 0;
    const save = vi.fn(async (_value: unknown): Promise<void> => undefined);
    const storage = browserDocumentStorage(() => ({ ...files }), value => { files = value; }, () => false);
    const services = await createServices({
      settings: { load: async () => null, save }, local: { get: () => null, set() {} }, host: host(),
      documents: storage,
      newId: () => `entity-${++sequence}`, now: () => '2026-09-22T12:00:00.000Z',
      scheduler: { after(ms, callback) { const timer = setTimeout(callback, ms); return () => clearTimeout(timer); } },
    });
    const features = createNoteFeatures({ storage, codec: markdownCodec, events: services.events, errors: services.diagnostics,
      newId: services.newId, now: () => '2026-09-23T00:00:00.000Z' }, register => ({
      task: register(taskFeature, () => services.preferences.current.taskFolder), project: register(projectFeature),
    }));
    try {
      const task = await features.repositories.task.create({ title: 'A Task' }, 'task-request');
      const project = await features.repositories.project.create({ name: 'A Project', budget: 0, archived: false }, 'project-request');
      if (!task.ok || !project.ok) throw new Error('Creation failed');
      expect(task.value.path.startsWith('Tasks/')).toBe(true); expect(project.value.path.startsWith('Projects/')).toBe(true);
      expect(project.value.values).toEqual({ name: 'A Project', budget: 0, archived: false });
      expect(files[project.value.path]).toContain('budget: 0\narchived: false\n');
      expect(save).not.toHaveBeenCalled();
      const updated = await features.repositories.project.update(project.value, { name: 'Revised', budget: 10, archived: true });
      if (!updated.ok) throw new Error('Update failed');
      expect(updated.value.id).toBe(project.value.id); expect(updated.value.path).toBe(project.value.path);
      expect((await features.repositories.project.delete(updated.value)).ok).toBe(true);
      expect(await features.repositories.project.list()).toEqual({ ok: true, value: [] });
      const tasks = await features.repositories.task.list(); expect(tasks.ok && tasks.value.length).toBe(1);
      await services.preferences.update({ taskFolder: 'Work/Tasks' });
      expect(await features.repositories.task.list()).toEqual({ ok: true, value: [] });
      const next = await features.repositories.task.create({ title: 'Moved folder' }, 'next-request');
      expect(next.ok && next.value.path.startsWith('Work/Tasks/')).toBe(true);
      expect(files[task.value.path]).toBeDefined();
    } finally { features.dispose(); services.dispose(); }
    expect(await features.repositories.task.list()).toMatchObject({ ok: false, error: { code: 'disposed' } });
    expect(await features.repositories.project.list()).toMatchObject({ ok: false, error: { code: 'disposed' } });
  });
});
