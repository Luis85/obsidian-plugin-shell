import type { ProjectRepository } from '../../application/ports/project-repository'
import type { EditorProject } from '../../domain/shared/project'
export class BrowserProjectRepository implements ProjectRepository {
 constructor(private readonly key='companion.vue-ir.v3'){}
 async load(){if(!import.meta.client)return null;const raw=localStorage.getItem(this.key);return raw?JSON.parse(raw):null}
 async save(project:EditorProject){if(import.meta.client)localStorage.setItem(this.key,JSON.stringify(project))}
}
