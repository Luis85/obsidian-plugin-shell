import { test, expect } from '@playwright/test';
import AxeBuilder from '@axe-core/playwright';
import '../../harness/app/test-api';
test('foundation loads without example data and retains accessible preferences and independent views', async ({ page }) => {
  const errors: string[] = []; page.on('pageerror', error => errors.push(error.message));
  await page.goto('/harness/app/'); await expect(page.locator('html')).toHaveAttribute('data-ready', 'true');
  await expect(page.getByTestId('foundation')).toBeVisible();
  await expect(page.getByRole('heading', { name: 'Your plugin', exact: true })).toBeVisible();
  for (const theme of ['light', 'dark']) {
    await page.locator('#theme-' + theme).click();
    await expect(page.locator('[data-plugin-ui]')).toHaveClass(new RegExp(theme));
    // Wait for Nuxt's foreground transition to reach the actual host token before auditing the stable view.
    await page.waitForFunction(() => {
      const root = document.querySelector('[data-plugin-ui]'); const action = document.querySelector('[aria-label="View actions"]');
      return !!root && !!action && getComputedStyle(root).color === getComputedStyle(action).color;
    });
    expect((await new AxeBuilder({ page }).include('[data-plugin-ui]').withTags(['wcag2a', 'wcag2aa', 'wcag21aa']).analyze()).violations).toEqual([]);
  }
  await page.getByLabel('Language', { exact: true }).selectOption('de');
  await page.getByRole('button', { name: 'Save preferences', exact: true }).click();
  await expect(page.getByRole('button', { name: 'Einstellungen speichern', exact: true })).toBeVisible();
  await page.evaluate(() => window.__SHELL_TEST__.mountSecond()); await expect(page.getByTestId('foundation')).toHaveCount(2);
  await page.evaluate(() => window.__SHELL_TEST__.closeSecond()); await expect(page.getByTestId('foundation')).toHaveCount(1);
  await page.reload(); await expect(page.locator('html')).toHaveAttribute('data-ready', 'true');
  await expect(page.getByRole('button', { name: 'Einstellungen speichern', exact: true })).toBeVisible();
  expect(await page.evaluate(() => window.__SHELL_TEST__.files())).toEqual({});
  expect(await page.evaluate(() => window.__SHELL_TEST__.faults)).toEqual([]); expect(errors).toEqual([]);
});
test('registered note features create distinct persisted notes through their real rendered actions', async ({ page }, testInfo) => {
  const errors: string[] = []; page.on('pageerror', error => errors.push(error.message));
  await page.goto('/harness/app/'); await expect(page.locator('html')).toHaveAttribute('data-ready', 'true');
  const panels = page.locator('section[class^="shell-authoring-"]'); let exercised = 0;
  for (const panel of await panels.all()) {
    const create = panel.getByRole('button', { name: 'Create note', exact: true });
    if (!await create.count()) continue;
    const before = Object.keys(await page.evaluate(() => window.__SHELL_TEST__.files())).length;
    await panel.getByRole('textbox').fill('Consumer first ' + exercised); await create.click();
    await expect.poll(async () => Object.keys(await page.evaluate(() => window.__SHELL_TEST__.files())).length).toBe(before + 1);
    await expect(panel.getByRole('status')).toHaveText('Note created');
    await panel.getByRole('textbox').fill('Consumer second ' + exercised); await create.click();
    await expect.poll(async () => Object.keys(await page.evaluate(() => window.__SHELL_TEST__.files())).length).toBe(before + 2);
    await expect(panel.getByRole('status')).toHaveText('Note created');
    const files = await page.evaluate(() => window.__SHELL_TEST__.files());
    expect(Object.keys(files)).toHaveLength(before + 2);
    expect(Object.keys(files).some(path => path.endsWith('/Consumer first ' + exercised + '.md'))).toBe(true);
    expect(Object.keys(files).some(path => path.endsWith('/Consumer second ' + exercised + '.md'))).toBe(true);
    expect(Object.values(files).some(text => text.includes('Consumer first ' + exercised))).toBe(true);
    expect(Object.values(files).some(text => text.includes('Consumer second ' + exercised))).toBe(true);
    await panel.getByRole('textbox').fill('Consumer first ' + exercised); await create.click();
    await expect(panel.getByRole('status')).toContainText('already exists');
    expect(await page.evaluate(() => window.__SHELL_TEST__.files())).toEqual(files);
    exercised++;
  }
  await testInfo.attach('consumer-note-feature-count', { body: JSON.stringify({ exercised }), contentType: 'application/json' });
  expect(await page.evaluate(() => window.__SHELL_TEST__.faults)).toEqual([]); expect(errors).toEqual([]);
});
